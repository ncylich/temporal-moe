      | out="$1"; shift
      | "$@" > "$out.stdout" 2> "$out.stderr" &
      | pid=$!
      | hwm=0; swp=0; rss=0

      |   [ -n "$w" ] && [ "$w" -gt "$swp" ] && swp=$w

      |   cat /proc/$pid/io > "$out.io" 2>/dev/null
      |   sleep 0.25
      | done
      | wait "$pid"; rc=$?

      | exit $rc
